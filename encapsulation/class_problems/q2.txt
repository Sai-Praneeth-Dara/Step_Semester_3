interface Printable {
    String printLabel();
}

class PackageBox implements Printable {

    private String trackingId;

    public PackageBox(String trackingId) {
        this.trackingId = trackingId;
    }

    @Override
    public String printLabel() {
        return "Package Label - Tracking ID: " + trackingId;
    }
}

class Invoice implements Printable {

    private String invoiceNumber;

    public Invoice(String invoiceNumber) {
        this.invoiceNumber = invoiceNumber;
    }

    @Override
    public String printLabel() {
        return "Invoice Label - Invoice Number: " + invoiceNumber;
    }
}

public class Main {

    static void printAll(Printable[] items) {

        for (Printable item : items) {
            System.out.println(item.printLabel());
        }
    }

    public static void main(String[] args) {

        Printable[] items = {
            new PackageBox("PKG101"),
            new Invoice("INV202"),
            new PackageBox("PKG303"),
            new Invoice("INV404")
        };

        printAll(items);
    }
}
