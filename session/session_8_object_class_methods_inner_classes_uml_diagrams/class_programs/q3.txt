import java.util.*;

abstract class Question {
    protected String questionId;
    protected String questionText;
    protected int points;

    public Question(String questionId, String questionText, int points) {
        this.questionId = questionId;
        this.questionText = questionText;
        this.points = points;
    }

    public String getQuestionId() {
        return questionId;
    }

    public int getPoints() {
        return points;
    }

    public abstract boolean evaluate(String answer);
}

class MultipleChoiceQuestion extends Question {
    private String correctOption;

    public MultipleChoiceQuestion(String id, String text, int points, String correctOption) {
        super(id, text, points);
        this.correctOption = correctOption;
    }

    @Override
    public boolean evaluate(String answer) {
        return answer != null && answer.equalsIgnoreCase(correctOption);
    }
}

class TrueFalseQuestion extends Question {
    private boolean correctAnswer;

    public TrueFalseQuestion(String id, String text, int points, boolean correctAnswer) {
        super(id, text, points);
        this.correctAnswer = correctAnswer;
    }

    @Override
    public boolean evaluate(String answer) {
        return answer != null && answer.equalsIgnoreCase(String.valueOf(correctAnswer));
    }
}

class ShortAnswerQuestion extends Question {
    private String correctAnswer;

    public ShortAnswerQuestion(String id, String text, int points, String correctAnswer) {
        super(id, text, points);
        this.correctAnswer = correctAnswer;
    }

    @Override
    public boolean evaluate(String answer) {
        return answer != null && answer.trim().equalsIgnoreCase(correctAnswer.trim());
    }
}

class Student {
    private String studentId;

    public Student(String studentId) {
        this.studentId = studentId;
    }

    public String getStudentId() {
        return studentId;
    }
}

class Examination {
    private String examName;
    private List<Question> questions = new ArrayList<>();

    public Examination(String examName) {
        this.examName = examName;
    }

    public String getExamName() {
        return examName;
    }

    public void addQuestion(Question question) {
        questions.add(question);
    }

    public List<Question> getQuestions() {
        return Collections.unmodifiableList(questions);
    }

    public int getTotalPoints() {
        int total = 0;
        for (Question question : questions) {
            total += question.getPoints();
        }
        return total;
    }
}

class Answer {
    private Question question;
    private String response;

    public Answer(Question question, String response) {
        this.question = question;
        this.response = response;
    }

    public Question getQuestion() {
        return question;
    }

    public String getResponse() {
        return response;
    }
}

class Attempt {
    private Student student;
    private Examination examination;
    private Map<String, Answer> answers = new LinkedHashMap<>();
    private boolean submitted;
    private int score;

    public Attempt(Student student, Examination examination) {
        this.student = student;
        this.examination = examination;
        this.submitted = false;
    }

    public boolean recordAnswer(String questionId, String response) {
        if (submitted) {
            System.out.println("Cannot change answers for a submitted examination.");
            return false;
        }

        for (Question question : examination.getQuestions()) {
            if (question.getQuestionId().equals(questionId)) {
                answers.put(questionId, new Answer(question, response));
                System.out.println("Answer recorded for " + questionId + ".");
                return true;
            }
        }

        System.out.println("Question not found: " + questionId);
        return false;
    }

    public void submit() {
        if (submitted) {
            return;
        }

        submitted = true;
        score = 0;

        System.out.println(examination.getExamName() + " submitted by "
                + student.getStudentId() + ".");

        for (Question question : examination.getQuestions()) {
            Answer answer = answers.get(question.getQuestionId());

            if (answer != null && question.evaluate(answer.getResponse())) {
                score += question.getPoints();
                System.out.println("Result: " + question.getQuestionId()
                        + ": Correct (" + question.getPoints() + " points)");
            } else {
                System.out.println("Result: " + question.getQuestionId()
                        + ": Incorrect (0 points)");
            }
        }

        System.out.println("Total score: " + score + "/"
                + examination.getTotalPoints());
    }

    public boolean isSubmitted() {
        return submitted;
    }

    public int getScore() {
        return score;
    }
}

class ExaminationSystem {
    private Map<String, Attempt> attempts = new HashMap<>();

    public Attempt startExam(Student student, Examination examination) {
        String key = student.getStudentId() + "-" + examination.getExamName();

        Attempt existing = attempts.get(key);

        if (existing != null && existing.isSubmitted()) {
            System.out.println("Student has already submitted this examination.");
            return null;
        }

        if (existing != null) {
            return existing;
        }

        Attempt attempt = new Attempt(student, examination);
        attempts.put(key, attempt);

        System.out.println(examination.getExamName() + " started by "
                + student.getStudentId() + ".");
        return attempt;
    }
}

public class Main {
    public static void main(String[] args) {
        Student student = new Student("Student 1");

        Examination exam = new Examination("Exam A");
        exam.addQuestion(new MultipleChoiceQuestion(
                "Question 1", "Which option is correct?", 5, "C"));
        exam.addQuestion(new TrueFalseQuestion(
                "Question 2", "Java is a procedural-only language.", 5, false));

        ExaminationSystem system = new ExaminationSystem();

        Attempt attempt = system.startExam(student, exam);

        if (attempt != null) {
            attempt.recordAnswer("Question 1", "C");
            attempt.recordAnswer("Question 2", "True");
            attempt.submit();
            attempt.recordAnswer("Question 1", "A");
        }
    }
}
