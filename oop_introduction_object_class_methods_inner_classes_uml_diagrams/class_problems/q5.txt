from abc import ABC, abstractmethod


class Staff(ABC):
    def __init__(self, name):
        self.name = name

    @abstractmethod
    def calculate_pay(self):
        pass

    @abstractmethod
    def get_type(self):
        pass

    def payslip(self):
        return f"Payslip[name={self.name}, type={self.get_type()}, pay={self.calculate_pay():g}]"


class FullTime(Staff):
    def __init__(self, name, salary):
        super().__init__(name)
        self.salary = salary

    def calculate_pay(self):
        return self.salary

    def get_type(self):
        return "FullTime"


class PartTime(Staff):
    def __init__(self, name, hours, rate):
        super().__init__(name)
        self.hours = hours
        self.rate = rate

    def calculate_pay(self):
        return self.hours * self.rate

    def get_type(self):
        return "PartTime"


class Intern(Staff):
    def __init__(self, name, stipend):
        super().__init__(name)
        self.stipend = stipend

    def calculate_pay(self):
        return self.stipend

    def get_type(self):
        return "Intern"


staff_members = []

while True:
    try:
        line = input().strip()
        if not line:
            break

        parts = line.split()
        staff_type = parts[0]
        name = parts[1]

        if staff_type == "FullTime":
            staff_members.append(FullTime(name, float(parts[2])))
        elif staff_type == "PartTime":
            staff_members.append(
                PartTime(name, float(parts[2]), float(parts[3]))
            )
        elif staff_type == "Intern":
            staff_members.append(Intern(name, float(parts[2])))

    except EOFError:
        break

total_pay = 0
top_earner = None

for staff in staff_members:
    print(staff.payslip())
    pay = staff.calculate_pay()
    total_pay += pay

    if top_earner is None or pay > top_earner.calculate_pay():
        top_earner = staff

print(f"Total {total_pay:g}")

if top_earner is not None:
    print(f"top earner {top_earner.name}")
