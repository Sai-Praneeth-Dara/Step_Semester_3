import java.util.Scanner;

class LibraryBook {
    String title;
    String isbn;

    // Main constructor
    LibraryBook(String title, String isbn) {
        this.title = title;
        this.isbn = isbn;
    }

    // Constructor for books without confirmed ISBN
    LibraryBook(String title) {
        this(title, "PENDING");
    }

    // Print book status
    void printStatus() {
        System.out.println("Book: " + title + " | ISBN: " + isbn);
    }
}

public class Main {
    public static void main(String[] args) {

        Scanner sc = new Scanner(System.in);

        System.out.print("Enter number of books: ");
        int n = sc.nextInt();
        sc.nextLine();

        String[] titles = new String[n];
        String[] isbns = new String[n];

        // Input
        for (int i = 0; i < n; i++) {
            System.out.print("Enter book title: ");
            titles[i] = sc.nextLine();

            System.out.print("Enter ISBN (leave blank if not confirmed): ");
            isbns[i] = sc.nextLine();
        }

        System.out.println("\nLibrary Book Status:");

        // Single pass processing
        for (int i = 0; i < n; i++) {

            LibraryBook book;

            if (isbns[i].isEmpty()) {
                book = new LibraryBook(titles[i]);
            } else {
                book = new LibraryBook(titles[i], isbns[i]);
            }

            book.printStatus();
        }

        sc.close();
    }
}
